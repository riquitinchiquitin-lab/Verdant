FROM node:24-slim AS builder

WORKDIR /app

COPY package*.json ./
RUN npm ci --legacy-peer-deps --no-audit --no-fund

COPY . .
RUN npm run build

FROM node:24-slim

WORKDIR /app

# Install curl for healthcheck and clean apt cache
RUN apt-get update && apt-get install -y --no-install-recommends curl \
    && rm -rf /var/lib/apt/lists/*

COPY package*.json ./
RUN npm ci --omit=dev --legacy-peer-deps --no-audit --no-fund \
    && npm cache clean --force \
    && rm -rf /root/.npm

COPY --from=builder /app/dist ./dist
COPY server.ts types.ts constants.ts tsconfig.json ./
COPY aggregator ./aggregator
COPY services ./services
COPY scripts ./scripts
COPY public ./public

# Setup data directory and drop root privileges
RUN mkdir -p /app/data && chmod 700 /app/data && chown -R node:node /app

USER node

EXPOSE 3000
ENV NODE_ENV=production

# Use the start script defined in package.json
CMD ["npm", "start"]
